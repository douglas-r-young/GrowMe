# GrowMe V0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working end-to-end demo of GrowMe V0 in ~24 hours: a Streamlit wizard that takes Linda's inputs, runs an agentic LangGraph pipeline (research → design doc → sessions → materials → nudges → delta report), produces editable design doc, and on [BUILD ON MIRO] populates a real Miro board with all 4 sessions, post-poll commitments, nudges, and a behavior delta report. Must pass a recorded 8-10 minute demo.

**Architecture:** Multi-step Streamlit wizard (UI) + LangGraph orchestrator (agentic flow) + Miro REST API (artifact canvas) + Apify (research) + multi-LLM via LiteLLM/LangChain (Featherless cheap, Claude Sonnet for heavy reasoning). Single edit point at the design doc; all other artifacts are generated on Build. Design lives in Streamlit, materials live on Miro. Pickle-based session state persistence.

**Tech Stack:** Python 3.11, Streamlit, LangGraph, LangChain (for LiteLLM integration), Apify Python SDK, Miro REST API (`requests`), Pydantic 2, pytest, python-dotenv. No DB, no auth, no real email/Slack send.

# src/growme/schemas.py
"""Pydantic schemas for the GrowMe pipeline. All LLM outputs validate here."""
from typing import Literal, TypedDict

from pydantic import BaseModel, Field, field_validator


# === INPUTS ===

class WizardInputs(BaseModel):
    company_url: str
    company_alias: str
    audience_description: str
    selected_behavior_ids: list[str]
    program_length_sessions: int = 4
    session_duration_min: int = 60

    @field_validator("selected_behavior_ids")
    @classmethod
    def _three_behaviors(cls, v: list[str]) -> list[str]:
        if len(v) != 3:
            raise ValueError("Exactly 3 behaviors must be selected")
        return v


# === BEHAVIOR MENU ===

class ResearchQuestions(BaseModel):
    examples: str
    baselines: str
    objections: str
    proof_points: str


class BehaviorTemplate(BaseModel):
    id: str
    name: str
    description: str
    framework_origin: str
    research_questions: ResearchQuestions


# === RESEARCH OUTPUT ===

class CitedFact(BaseModel):
    text: str
    source: str
    confidence: Literal["high", "medium", "low"]


class BehaviorFindings(BaseModel):
    examples: list[CitedFact] = Field(default_factory=list, max_length=5)
    baselines: list[CitedFact] = Field(default_factory=list, max_length=3)
    objections: list[CitedFact] = Field(default_factory=list, max_length=5)
    proof_points: list[CitedFact] = Field(default_factory=list, max_length=5)


class BehaviorContext(BaseModel):
    behavior_id: str
    behavior_name: str
    behavior_description: str
    framework_origin: str
    research_questions: list[str]
    findings: BehaviorFindings


class BaseCompanyResearch(BaseModel):
    company_snapshot: str
    customer_voice: list[CitedFact] = Field(default_factory=list)
    vertical_vocab: list[str] = Field(default_factory=list)
    named_competitors: list[CitedFact] = Field(default_factory=list)


class EnrichedContext(BaseModel):
    base: BaseCompanyResearch
    per_behavior: list[BehaviorContext]
    sources_used: list[str] = Field(default_factory=list)
    research_timestamp: str


# === DESIGN DOC ===

class DesignDoc(BaseModel):
    audience_section_md: str
    behavior_objectives: list[str]
    learning_objectives: list[str]
    full_markdown: str


# === SESSIONS + MATERIALS ===

class AgendaBlock(BaseModel):
    name: str
    duration_min: int
    description: str


class SessionPlan(BaseModel):
    session_number: int
    title: str
    behavior_id: str | None
    learning_objective: str
    agenda: list[AgendaBlock]


class SessionMaterials(BaseModel):
    session_number: int
    miro_frame_id: str
    slide_frame_ids: list[str]
    pre_poll_id: str
    post_poll_id: str
    facilitator_guide_doc_id: str


# === SIMULATED RUNTIME ===

class LearnerResponse(BaseModel):
    learner_id: str
    learner_name: str
    session_number: int
    pre_poll_answers: dict[str, str]
    post_poll_commitment: str
    nudge_replied: bool


class Nudge(BaseModel):
    learner_id: str
    session_number: int
    email_subject: str
    email_body_md: str
    slack_text: str
    proof_point_used: CitedFact
    miro_card_id: str


class BehaviorMovement(BaseModel):
    behavior_id: str
    pct_moved_from_rarely_to_often: float
    pre_distribution: dict[str, int]
    post_distribution: dict[str, int]


class DeltaReport(BaseModel):
    behavior_movements: list[BehaviorMovement]
    top_objection_still_surfacing: CitedFact
    recommended_reinforcement_md: str
    full_markdown: str
    miro_doc_id: str


# === ORCHESTRATOR STATE ===

class GrowMeState(TypedDict, total=False):
    wizard_inputs: WizardInputs
    miro_board_id: str

    enriched_context: EnrichedContext | None
    design_doc: DesignDoc | None
    design_doc_edited_md: str | None

    session_plans: list[SessionPlan] | None
    materials: list[SessionMaterials] | None

    learner_responses: list[LearnerResponse]
    nudges: list[Nudge]
    delta_report: DeltaReport | None
```

```python
# src/growme/llm_clients.py
"""LiteLLM-backed unified completion. Roles map to provider/model strings.

LiteLLM accepts model strings like:
  anthropic/claude-sonnet-4-6
  openai/gpt-4o
  featherless_ai/<model_name>
"""
from __future__ import annotations

import json
import os

import litellm
from pydantic import BaseModel

# Role -> model id (LiteLLM format: provider/model)
ROLE_TO_MODEL: dict[str, str] = {
    "research_extract": "featherless_ai/meta-llama/Meta-Llama-3.1-8B-Instruct",
    "research_synth":   "featherless_ai/meta-llama/Meta-Llama-3.1-70B-Instruct",
    "design_doc":       "featherless_ai/meta-llama/Meta-Llama-3.1-70B-Instruct",
    "session_plan":     "anthropic/claude-sonnet-4-6",
    "materials":        "anthropic/claude-sonnet-4-6",
    "nudges":           "featherless_ai/meta-llama/Meta-Llama-3.1-8B-Instruct",
    "delta_report":     "anthropic/claude-sonnet-4-6",
}


def resolve_model_for_role(role: str) -> str:
    return ROLE_TO_MODEL[role]


def complete(
    role: str,
    system: str,
    user: str,
    *,
    temperature: float = 0.3,
    max_tokens: int = 4096,
) -> str:
    """Single-shot completion. Returns the raw text content."""
    model = resolve_model_for_role(role)
    resp = litellm.completion(
        model=model,
        messages=[
            {"role": "system", "content": system},
            {"role": "user", "content": user},
        ],
        temperature=temperature,
        max_tokens=max_tokens,
        api_key=_api_key_for(model),
    )
    return resp.choices[0].message.content or ""


def complete_json(
    role: str,
    system: str,
    user: str,
    schema: type[BaseModel],
    *,
    temperature: float = 0.1,
    max_tokens: int = 4096,
    max_retries: int = 2,
) -> BaseModel:
    """Completion that must return JSON validating against `schema`. Retries with feedback."""
    schema_hint = json.dumps(schema.model_json_schema(), indent=2)
    full_system = (
        f"{system}\n\nReturn ONLY a JSON object matching this schema. "
        f"No prose, no markdown fences.\n\n{schema_hint}"
    )

    last_err: str | None = None
    for attempt in range(max_retries + 1):
        prompt = user if last_err is None else (
            f"{user}\n\nPrior attempt failed validation: {last_err}. "
            "Return strictly valid JSON for the schema above."
        )
        text = complete(
            role, full_system, prompt,
            temperature=temperature, max_tokens=max_tokens,
        )
        try:
            return schema.model_validate_json(_strip_fences(text))
        except Exception as e:
            last_err = str(e)
    raise ValueError(f"complete_json failed after {max_retries+1} attempts: {last_err}")


def _strip_fences(text: str) -> str:
    """Remove ```json ... ``` wrappers if a model adds them anyway."""
    t = text.strip()
    if t.startswith("```"):
        t = t.split("\n", 1)[1] if "\n" in t else t[3:]
        if t.endswith("```"):
            t = t[:-3]
    return t.strip()


def _api_key_for(model: str) -> str | None:
    if model.startswith("anthropic/"):
        return os.environ["ANTHROPIC_API_KEY"]
    if model.startswith("openai/"):
        return os.environ["OPENAI_API_KEY"]
    if model.startswith("featherless_ai/"):
        return os.environ["FEATHERLESS_API_KEY"]
    return None
```

# src/growme/miro_client.py
"""Thin Miro REST API wrapper. Server-side only (we don't use MCP at runtime)."""
from __future__ import annotations

import os
import time
from typing import Any

import requests

API_BASE = "https://api.miro.com/v2"
RATE_LIMIT_SLEEP = 0.2  # 5 calls/sec ceiling


class MiroClient:
    def __init__(self, token: str | None = None, board_id: str | None = None):
        self.token = token or os.environ["MIRO_TOKEN"]
        self.board_id = board_id or os.environ["MIRO_BOARD_ID"]
        self._headers = {
            "Authorization": f"Bearer {self.token}",
            "Content-Type": "application/json",
            "Accept": "application/json",
        }

    def _request(self, method: str, path: str, **kwargs) -> dict[str, Any]:
        time.sleep(RATE_LIMIT_SLEEP)
        url = f"{API_BASE}{path}"
        r = requests.request(method, url, headers=self._headers, timeout=30, **kwargs)
        r.raise_for_status()
        return r.json() if r.content else {}

    # --- frames ---
    def create_frame(self, title: str, x: float, y: float, width: int = 1200, height: int = 800) -> str:
        body = {
            "data": {"title": title, "format": "custom"},
            "geometry": {"width": width, "height": height},
            "position": {"x": x, "y": y},
        }
        resp = self._request("POST", f"/boards/{self.board_id}/frames", json=body)
        return resp["id"]

    # --- text on frames (used as slide content) ---
    def create_text(self, content_html: str, x: float, y: float, parent_frame_id: str | None = None,
                    width: int = 800) -> str:
        body: dict[str, Any] = {
            "data": {"content": content_html},
            "position": {"x": x, "y": y},
            "geometry": {"width": width},
        }
        if parent_frame_id:
            body["parent"] = {"id": parent_frame_id}
        resp = self._request("POST", f"/boards/{self.board_id}/texts", json=body)
        return resp["id"]

    # --- documents ---
    def create_document(self, title: str, content_md: str, x: float, y: float,
                        parent_frame_id: str | None = None) -> str:
        body: dict[str, Any] = {
            "data": {"title": title, "documentUrl": _md_to_data_url(content_md)},
            "position": {"x": x, "y": y},
        }
        if parent_frame_id:
            body["parent"] = {"id": parent_frame_id}
        resp = self._request("POST", f"/boards/{self.board_id}/documents", json=body)
        return resp["id"]

    # --- cards (used for nudges) ---
    def create_card(self, title: str, description_md: str, x: float, y: float,
                    parent_frame_id: str | None = None) -> str:
        body: dict[str, Any] = {
            "data": {"title": title, "description": description_md},
            "position": {"x": x, "y": y},
        }
        if parent_frame_id:
            body["parent"] = {"id": parent_frame_id}
        resp = self._request("POST", f"/boards/{self.board_id}/cards", json=body)
        return resp["id"]

    # --- sticky notes (used for poll responses) ---
    def create_sticky(self, content: str, x: float, y: float, color: str = "yellow",
                      parent_frame_id: str | None = None) -> str:
        body: dict[str, Any] = {
            "data": {"content": content, "shape": "square"},
            "style": {"fillColor": color},
            "position": {"x": x, "y": y},
        }
        if parent_frame_id:
            body["parent"] = {"id": parent_frame_id}
        resp = self._request("POST", f"/boards/{self.board_id}/sticky_notes", json=body)
        return resp["id"]

    # --- delete (used in dev/demo reset) ---
    def delete_item(self, item_id: str) -> None:
        try:
            self._request("DELETE", f"/boards/{self.board_id}/items/{item_id}")
        except requests.HTTPError:
            pass  # idempotent best-effort


def _md_to_data_url(md: str) -> str:
    """Miro documents accept a documentUrl. We host inline via base64 PDF — but for
    MVP we render markdown into a hosted URL is overkill. Use a data: URL with text/plain.
    The Miro UI will render whatever the URL serves; for hackathon a text/plain data URL
    is sufficient.

    NOTE: If Miro rejects data: URLs in your tier, fall back to creating a text widget
    with the markdown content instead.
    """
    import base64
    encoded = base64.b64encode(md.encode("utf-8")).decode("ascii")
    return f"data:text/plain;base64,{encoded}"
```

# src/growme/research/aliasing.py
"""Replace real company name + URL with an alias. Demo-safe wrapper for Photon DB ↔ Neon."""
from __future__ import annotations

import re
from typing import Any


def alias_text(text: str, real: str, alias: str, real_url: str, alias_url: str) -> str:
    """Case-insensitive replace of real → alias in any string."""
    out = re.sub(re.escape(real_url), alias_url, text, flags=re.IGNORECASE)
    out = re.sub(re.escape(real), alias, out, flags=re.IGNORECASE)
    return out


def alias_obj(obj: Any, real: str, alias: str, real_url: str, alias_url: str) -> Any:
    """Recursively alias every string inside a JSON-like nested structure."""
    if isinstance(obj, str):
        return alias_text(obj, real, alias, real_url, alias_url)
    if isinstance(obj, dict):
        return {k: alias_obj(v, real, alias, real_url, alias_url) for k, v in obj.items()}
    if isinstance(obj, list):
        return [alias_obj(v, real, alias, real_url, alias_url) for v in obj]
    return obj
```

# src/growme/app/state.py
"""Pickle-backed session state. One UUID per Streamlit session."""
from __future__ import annotations

import os
import pickle
import uuid
from pathlib import Path
from typing import Any

import streamlit as st


def _session_dir() -> Path:
    p = Path(os.environ.get("GROWME_SESSION_DIR", ".growme_sessions"))
    p.mkdir(exist_ok=True)
    return p


def session_uuid() -> str:
    if "session_uuid" not in st.session_state:
        st.session_state["session_uuid"] = str(uuid.uuid4())
    return st.session_state["session_uuid"]


def _path_for(uid: str) -> Path:
    return _session_dir() / f"{uid}.pkl"


def save(data: dict[str, Any]) -> None:
    with _path_for(session_uuid()).open("wb") as f:
        pickle.dump(data, f)


def load() -> dict[str, Any]:
    p = _path_for(session_uuid())
    if not p.exists():
        return {}
    with p.open("rb") as f:
        return pickle.load(f)


def update(key: str, value: Any) -> None:
    data = load()
    data[key] = value
    save(data)


def get(key: str, default: Any = None) -> Any:
    return load().get(key, default)
```

from __future__ import annotations

from concurrent.futures import ThreadPoolExecutor, as_completed

from growme.research.phase_a import (
    company_snapshot,
    customer_voice,
    named_competitors,
    vertical_vocab,

def run(company_url: str, company_alias: str, g2_product_url: str | None = None) -> BaseCompanyResearch:

        "snapshot": (company_snapshot.run, (company_url,)),
        "vocab":    (vertical_vocab.run, (company_url,)),
        "competitors": (named_competitors.run, (company_alias, company_url)),

    if g2_product_url:

        future_to_key = {ex.submit(fn, *args): key for key, (fn, args) in tasks.items()}

            key = future_to_key[fut]
            try:
                results[key] = fut.result()

                print(f"[phase_a] {key} failed: {e}")

    return BaseCompanyResearch(
        company_snapshot=results.get("snapshot") or "",
        customer_voice=results.get("voice") or [],
        vertical_vocab=results.get("vocab") or [],
        named_competitors=results.get("competitors") or [],

from __future__ import annotations

    source_url: str = ""


class _BucketResult(BaseModel):
    findings: list[_Finding] = Field(default_factory=list, max_length=5)

    "You are a sales-enablement researcher synthesizing into a specific finding bucket. "
    "Given web search hits and a question, return up to 5 concrete findings. "
    "Each finding has 'text' (1-2 sentence factual statement) and 'source_url' "
    "(URL from the search hits if directly supported; empty string if inferred). "
    "Ground claims in the hits whenever possible; do not fabricate URLs."
)


def _run_one_question(question: str, bucket: Bucket) -> list[CitedFact]:
    hits = search(question, max_results=5)
    blob = "\n".join(f"- [{h.title}]({h.url}): {h.content}" for h in hits) or "(no results)"
    parsed = complete_json(
        role="research_synth",
        system=SYNTH_SYSTEM,
        user=f"Question (bucket={bucket}):\n{question}\n\nWeb hits:\n{blob}",
        schema=_BucketResult,
    )
    out: list[CitedFact] = []
    for f in parsed.findings:
        if f.source_url:
            out.append(cited_fact(f.text, source_web(f.source_url), "high"))
        else:
            out.append(cited_fact(f.text, source_inference("research_synth"), "low"))
    return out

            ex.submit(_run_one_question, q, b): b
            for q, b in zip(questions, buckets, strict=True)
        }
        for fut in as_completed(future_to_bucket):
            bucket = future_to_bucket[fut]
            try:
                findings[bucket] = fut.result()
            except Exception as e:
                print(f"[phase_b/{behavior_id}/{bucket}] failed: {e}")
                findings[bucket] = []

    return BehaviorContext(
        behavior_id=behavior_id,
        behavior_name=template.name,
        behavior_description=template.description,
        framework_origin=template.framework_origin,
        research_questions=questions,
        findings=BehaviorFindings(
            examples=findings["examples"][:5],
            baselines=findings["baselines"][:3],
            objections=findings["objections"][:5],
            proof_points=findings["proof_points"][:5],
        ),
    )

Usage from LangGraph:

import datetime
import json
import os

from growme.research.phase_a.orchestrator import run as run_phase_a

G2_URL = "https://www.g2.com/products/neon/reviews"

FIXTURES_DIR = Path(__file__).parent / "fixtures"


def run(inputs: WizardInputs) -> EnrichedContext:
    """Run live or load from fixture cache based on USE_LIVE_RESEARCH."""
    use_live = os.environ.get("USE_LIVE_RESEARCH", "true").lower() == "true"
    if not use_live:
        cached = _try_load_fixture(inputs)
        if cached is not None:
            return cached

    base_future_args = (REAL_URL, REAL_COMPANY, G2_URL)
    behavior_args = inputs.selected_behavior_ids

    with ThreadPoolExecutor(max_workers=4) as ex:
        base_future = ex.submit(run_phase_a, *base_future_args)
        beh_futures = {ex.submit(run_phase_b, bid, REAL_COMPANY): bid for bid in behavior_args}

        base = base_future.result()
        per_behavior_by_id: dict[str, object] = {}
        for fut in as_completed(beh_futures):
            bid = beh_futures[fut]
            per_behavior_by_id[bid] = fut.result()

    # Preserve user-selected order
    per_behavior = [per_behavior_by_id[bid] for bid in behavior_args]

    sources: list[str] = []
    for cf in base.customer_voice + base.named_competitors:
        sources.append(cf.source)
    for bc in per_behavior:
        for bucket in (bc.findings.examples, bc.findings.baselines,
                       bc.findings.objections, bc.findings.proof_points):
            for cf in bucket:
                sources.append(cf.source)

    enriched = EnrichedContext(
        base=base,
        per_behavior=per_behavior,
        sources_used=sorted(set(sources)),
        research_timestamp=datetime.datetime.utcnow().isoformat() + "Z",
    )

    # Alias the whole tree (Neon → Photon DB / neon.tech → photondb.io)
    aliased_dict = alias_obj(
        enriched.model_dump(),
        REAL_COMPANY, inputs.company_alias,
        REAL_URL, _alias_url(inputs.company_alias),
    )

    _write_fixture(inputs, aliased)
    return aliased


def _alias_url(alias: str) -> str:
    """Cheap alias-URL guess: lowercase + '.io'. Override per project if needed."""

DELTA_X, DELTA_Y = -2400, 2200


def create_session_frame(client: MiroClient, plan: SessionPlan) -> str:
    """One outer frame per session, ordered left to right by session_number."""
    x = (plan.session_number - 1) * SESSION_X_STEP
    return client.create_frame(
        title=f"Session {plan.session_number}: {plan.title}",
        x=x,
        y=SESSION_Y,
        width=1300,
        height=1100,
    )


def write_slide(client: MiroClient, frame_id: str, slide_idx: int, title: str, body_md: str) -> str:
    """Create a slide as a text widget inside the session frame."""
    y_offset = -400 + slide_idx * 130
    content = f"<h3>{title}</h3><p>{body_md}</p>"
    return client.create_text(content, x=-200, y=y_offset, parent_frame_id=frame_id, width=600)

    """Pre-poll baseline. Render as a text widget with the question + options."""
    options_html = "<br>".join(f"☐ {o}" for o in options)
    content = f"<h4>Pre-poll: {question}</h4><p>{options_html}</p>"
    return client.create_text(content, x=400, y=-300, parent_frame_id=frame_id, width=400)


def write_post_poll(client: MiroClient, frame_id: str, prompt: str, options: list[str]) -> str:
    """Post-session commitment poll."""
    options_html = "<br>".join(f"☐ {o}" for o in options)
    content = f"<h4>Commitment poll: {prompt}</h4><p>{options_html}</p>"
    return client.create_text(content, x=400, y=100, parent_frame_id=frame_id, width=400)

        x=400,
        y=400,
        parent_frame_id=frame_id,

"""Generate materials (slides, polls, facilitator guide) and write to Miro.

Per session: spawns 3 LLM calls in parallel (slides, polls, guide), then writes them
to the session's Miro frame.
"""
from __future__ import annotations

from concurrent.futures import ThreadPoolExecutor
from typing import Any

from pydantic import BaseModel, Field

from growme.llm_clients import complete_json
from growme.miro_client import MiroClient
from growme.schemas import BehaviorContext, EnrichedContext, SessionMaterials, SessionPlan
from growme.sessions import miro_writers


# --- Schemas the LLM must produce ---

class _Slide(BaseModel):
    title: str
    body_md: str


class _SlidesBundle(BaseModel):
    slides: list[_Slide] = Field(min_length=5, max_length=8)


class _Poll(BaseModel):
    question: str
    options: list[str] = Field(min_length=3, max_length=5)


class _PollsBundle(BaseModel):
    pre_poll: _Poll
    post_commitment_poll: _Poll


class _Guide(BaseModel):
    full_markdown: str


# --- Prompts ---

SLIDES_SYSTEM = """\
You are an instructional designer building slides for a 60-minute training session.
Produce 5-8 slides, each with a short title and 2-4 bullets of body content (markdown).
Ground content in the provided behavior findings. Cite specific examples and objections
where relevant. Output strict JSON: { "slides": [Slide, ...] }.
"""

POLLS_SYSTEM = """\
You produce two polls for one training session:
  pre_poll: a baseline frequency question matching this behavior, options like
    "Never", "Rarely", "Sometimes", "Often", "Always".
  post_commitment_poll: a single-select commitment question with 4-5 concrete options,
    each option phrased as a specific action the learner could take in the next 7 days.
Output strict JSON: { "pre_poll": Poll, "post_commitment_poll": Poll }.
"""

GUIDE_SYSTEM = """\
You produce a markdown facilitator guide for a 60-minute training session.
Sections (use level-2 headings):
  ## Setup (10 min before)
  ## Opening hook (5 min)
  ## Teach (15 min)  -- include specific points + how to handle the top 1-2 objections
  ## Discuss (15 min) -- 2-3 discussion prompts
  ## Practice / Role-play (15 min) -- include a fully-spelled-out scenario
  ## Commitment + close (10 min)
Ground objections + scenarios in the provided behavior findings.
Output strict JSON: { "full_markdown": "..." }.
"""


def _format_behavior_findings(bc: BehaviorContext) -> str:
    def fmt(label, items):
        if not items:
            return f"{label}: (none)"
        return f"{label}:\n" + "\n".join(f"- {f.text}" for f in items[:5])
    return (
        f"BEHAVIOR: {bc.behavior_name} ({bc.framework_origin})\n"
        f"DESCRIPTION: {bc.behavior_description}\n\n"
        f"{fmt('EXAMPLES', bc.findings.examples)}\n\n"
        f"{fmt('BASELINES', bc.findings.baselines)}\n\n"
        f"{fmt('OBJECTIONS', bc.findings.objections)}\n\n"
        f"{fmt('PROOF_POINTS', bc.findings.proof_points)}\n"
    )


def _format_integration_findings(enriched: EnrichedContext) -> str:
    """Combined view of all 3 behaviors for the integration session."""
    blocks = [_format_behavior_findings(bc) for bc in enriched.per_behavior]
    return "\n\n---\n\n".join(blocks)


def _gen_slides(plan: SessionPlan, behavior_blob: str) -> _SlidesBundle:
    user = (
        f"SESSION: {plan.title}\n"
        f"LEARNING OBJECTIVE: {plan.learning_objective}\n\n"
        f"FINDINGS:\n{behavior_blob}\n"
    )
    return complete_json(role="materials", system=SLIDES_SYSTEM, user=user, schema=_SlidesBundle)


def _gen_polls(plan: SessionPlan, behavior_blob: str) -> _PollsBundle:
    user = (
        f"SESSION: {plan.title}\n"
        f"LEARNING OBJECTIVE: {plan.learning_objective}\n\n"
        f"FINDINGS:\n{behavior_blob}\n"
    )
    return complete_json(role="materials", system=POLLS_SYSTEM, user=user, schema=_PollsBundle)


def _gen_guide(plan: SessionPlan, behavior_blob: str) -> _Guide:
    user = (
        f"SESSION: {plan.title}\n"
        f"LEARNING OBJECTIVE: {plan.learning_objective}\n\n"
        f"AGENDA:\n" + "\n".join(f"- {a.name} ({a.duration_min}m): {a.description}" for a in plan.agenda)
        + "\n\nFINDINGS:\n" + behavior_blob
    )
    return complete_json(role="materials", system=GUIDE_SYSTEM, user=user, schema=_Guide,
                         max_tokens=3000)


def run_for_session(
    client: MiroClient,
    plan: SessionPlan,
    enriched: EnrichedContext,
) -> SessionMaterials:
    if plan.behavior_id is None:
        behavior_blob = _format_integration_findings(enriched)
    else:
        bc = next(b for b in enriched.per_behavior if b.behavior_id == plan.behavior_id)
        behavior_blob = _format_behavior_findings(bc)

    # Parallel LLM calls per session
    with ThreadPoolExecutor(max_workers=3) as ex:
        f_slides = ex.submit(_gen_slides, plan, behavior_blob)
        f_polls  = ex.submit(_gen_polls, plan, behavior_blob)
        f_guide  = ex.submit(_gen_guide, plan, behavior_blob)
        slides_bundle = f_slides.result()
        polls_bundle  = f_polls.result()
        guide         = f_guide.result()

    # Sequential Miro writes (rate limit)
    frame_id = miro_writers.create_session_frame(client, plan)
    slide_ids = [
        miro_writers.write_slide(client, frame_id, i, s.title, s.body_md)
        for i, s in enumerate(slides_bundle.slides)
    ]
    pre_poll_id = miro_writers.write_pre_poll(client, frame_id, polls_bundle.pre_poll.question,
                                              polls_bundle.pre_poll.options)
    post_poll_id = miro_writers.write_post_poll(client, frame_id,
                                                polls_bundle.post_commitment_poll.question,
                                                polls_bundle.post_commitment_poll.options)
    guide_doc_id = miro_writers.write_facilitator_guide(client, frame_id, plan.session_number,
                                                        guide.full_markdown)

    return miro_writers.materials_skeleton(
        plan=plan,
        miro_frame_id=frame_id,
        slide_ids=slide_ids,
        pre_poll_id=pre_poll_id,
        post_poll_id=post_poll_id,
        guide_doc_id=guide_doc_id,
    )


def run_all(
    client: MiroClient,
    plans: list[SessionPlan],
    enriched: EnrichedContext,
) -> list[SessionMaterials]:
    """Sequential per-session iteration to keep Miro calls within rate limit.
    LLM calls inside each session are parallel (3-way).
    """
    out: list[SessionMaterials] = []
    for p in plans:
        out.append(run_for_session(client, p, enriched))
    return out

# src/growme/nudges/node.py
"""Generate per-learner nudges for a session and write each to a Miro card."""
from __future__ import annotations

import random

from pydantic import BaseModel, Field

from growme.llm_clients import complete_json
from growme.miro_client import MiroClient
from growme.nudges.learner_fixtures import LEARNERS
from growme.schemas import BehaviorContext, CitedFact, EnrichedContext, LearnerResponse, Nudge

NUDGES_X = -2400
NUDGES_Y = 1400


class _NudgePair(BaseModel):
    email_subject: str
    email_body_md: str = Field(min_length=40)
    slack_text: str = Field(min_length=20)


NUDGE_SYSTEM = """\
You write a one-week post-training nudge for a sales rep.
Inputs: their COMMITMENT (a single concrete action they pledged), and a relevant
PROOF POINT (a fact about the company's product they should know).

Output strict JSON with:
  email_subject: short, second-person ("Your X commitment...")
  email_body_md: 60-100 words, encouraging, concrete; cite the proof point naturally
  slack_text: 20-40 words, casual, prompt them to reply yes/no on whether they did the action
"""


def _pick_proof_point(bc: BehaviorContext) -> CitedFact | None:
    if bc.findings.proof_points:
        return random.choice(bc.findings.proof_points)
    return None


def _make_nudge_text(commitment: str, proof_point_text: str) -> _NudgePair:
    user = (
        f"COMMITMENT: {commitment}\n\n"
        f"PROOF POINT: {proof_point_text}\n"
    )
    return complete_json(role="nudges", system=NUDGE_SYSTEM, user=user, schema=_NudgePair)


def _frame_for_session(client: MiroClient, session_number: int) -> str:
    """Create or reuse a per-session 'Nudges S<N>' frame."""
    return client.create_frame(
        title=f"Nudges — Session {session_number}",
        x=NUDGES_X,
        y=NUDGES_Y + (session_number - 1) * 900,
        width=1300,
        height=800,
    )


def generate_for_session(
    client: MiroClient,
    session_number: int,
    all_responses: list[LearnerResponse],
    enriched: EnrichedContext,
) -> list[Nudge]:
    """Generate nudges for all learners who completed `session_number`."""
    if session_number == 4 or len(enriched.per_behavior) < session_number:
        bc = enriched.per_behavior[0]  # integration session pulls from any behavior
    else:
        bc = enriched.per_behavior[session_number - 1]

    frame_id = _frame_for_session(client, session_number)

    out: list[Nudge] = []
    learners_for_session = [r for r in all_responses if r.session_number == session_number]

    for i, r in enumerate(learners_for_session):
        proof = _pick_proof_point(bc)
        proof_text = proof.text if proof else "Use a recent customer outcome to anchor your message."
        pair = _make_nudge_text(r.post_poll_commitment, proof_text)
        title = f"{r.learner_name}: {pair.email_subject}"
        body = (
            f"**Email:**\n\n{pair.email_body_md}\n\n---\n\n"
            f"**Slack:**\n\n{pair.slack_text}\n\n---\n\n"
            f"_Proof point used:_ {proof_text}"
        )
        x = -550 + (i % 4) * 280
        y = -300 + (i // 4) * 350
        card_id = client.create_card(title, body, x=x, y=y, parent_frame_id=frame_id)
        out.append(Nudge(
            learner_id=r.learner_id,
            session_number=session_number,
            email_subject=pair.email_subject,
            email_body_md=pair.email_body_md,
            slack_text=pair.slack_text,
            proof_point_used=proof or CitedFact(
                text=proof_text, source="inference:nudges", confidence="low"
            ),
            miro_card_id=card_id,
        ))
    return out

# src/growme/delta_report/node.py
"""Generate a behavior delta report from pre/post fixture distributions + LLM reasoning."""
from __future__ import annotations

from pydantic import BaseModel

from growme.llm_clients import complete_json
from growme.miro_client import MiroClient
from growme.nudges import learner_fixtures
from growme.schemas import (
    BehaviorMovement,
    CitedFact,
    DeltaReport,
    EnrichedContext,
    LearnerResponse,
)

DELTA_X, DELTA_Y = -2400, 5000


class _DeltaReasoning(BaseModel):
    top_objection_still_surfacing_text: str
    top_objection_source: str = ""
    recommended_reinforcement_md: str
    summary_md: str


DELTA_SYSTEM = """\
You are an L&D analyst. Given pre/post poll distributions per session and a list of
behavior findings (especially objections), return:
  - top_objection_still_surfacing_text: the single most likely objection that learners
    will still face given the residual 'rarely/sometimes' counts
  - top_objection_source: a URL from the provided findings if one applies, else ""
  - recommended_reinforcement_md: 2-4 bullets recommending follow-up reinforcement,
    grounded in the proof_points provided
  - summary_md: a 200-word executive summary suitable for the program's L&D lead

Output strict JSON.
"""


def _movements(responses: list[LearnerResponse], behavior_id_for_session: dict[int, str | None]) -> list[BehaviorMovement]:
    out: list[BehaviorMovement] = []
    for s in (1, 2, 3):
        bid = behavior_id_for_session[s]
        if bid is None:
            continue
        pre = learner_fixtures.pre_distribution(s)
        post = learner_fixtures.post_distribution(s)
        pre_often = pre.get("Often", 0) + pre.get("Always", 0)
        post_often = post.get("Often", 0) + post.get("Always", 0)
        pct = 0.0
        n = sum(pre.values()) or 1
        pct = (post_often - pre_often) / n
        out.append(BehaviorMovement(
            behavior_id=bid,
            pct_moved_from_rarely_to_often=round(pct, 3),
            pre_distribution=pre,
            post_distribution=post,
        ))
    return out


def run(
    all_responses: list[LearnerResponse],
    enriched: EnrichedContext,
    client: MiroClient,
) -> DeltaReport:
    behavior_id_for_session = {1: enriched.per_behavior[0].behavior_id,
                               2: enriched.per_behavior[1].behavior_id,
                               3: enriched.per_behavior[2].behavior_id,
                               4: None}
    movements = _movements(all_responses, behavior_id_for_session)

    findings_blob = "\n\n".join(
        f"### {bc.behavior_name}\nObjections:\n"
        + "\n".join(f"- {f.text} (src={f.source})" for f in bc.findings.objections[:5])
        + "\nProof points:\n"
        + "\n".join(f"- {f.text} (src={f.source})" for f in bc.findings.proof_points[:5])
        for bc in enriched.per_behavior
    )
    pre_post_blob = "\n".join(
        f"S{m.behavior_id}: pre={m.pre_distribution}, post={m.post_distribution}, "
        f"pct moved={m.pct_moved_from_rarely_to_often:+.0%}"
        for m in movements
    )

    reasoning = complete_json(
        role="delta_report",
        system=DELTA_SYSTEM,
        user=f"PRE/POST DISTRIBUTIONS:\n{pre_post_blob}\n\nFINDINGS:\n{findings_blob}",
        schema=_DeltaReasoning,
        max_tokens=2000,
    )

    full_md = (
        f"# Behavior Delta Report\n\n"
        f"## Movements\n\n"
        + "\n".join(
            f"- **{m.behavior_id}**: {m.pct_moved_from_rarely_to_often:+.0%} moved into Often+\n"
            f"  - pre: {m.pre_distribution}\n  - post: {m.post_distribution}"
            for m in movements
        )
        + f"\n\n## Top objection still surfacing\n\n"
        f"_{reasoning.top_objection_still_surfacing_text}_\n\n"
        f"## Recommended reinforcement\n\n{reasoning.recommended_reinforcement_md}\n\n"
        f"## Summary\n\n{reasoning.summary_md}\n"
    )

    frame_id = client.create_frame("Behavior Delta Report", x=DELTA_X, y=DELTA_Y,
                                   width=1500, height=1000)
    doc_id = client.create_document(
        title="Behavior Delta Report",
        content_md=full_md,
        x=0,
        y=0,
        parent_frame_id=frame_id,
    )

    return DeltaReport(
        behavior_movements=movements,
        top_objection_still_surfacing=CitedFact(
            text=reasoning.top_objection_still_surfacing_text,
            source=(reasoning.top_objection_source or "inference:delta_report"),
            confidence="medium" if reasoning.top_objection_source else "low",
        ),
        recommended_reinforcement_md=reasoning.recommended_reinforcement_md,
        full_markdown=full_md,
        miro_doc_id=doc_id,
    )
```

"""Programmatic end-to-end run for timing + smoke without the UI.

Times each phase; total target <3 minutes from research start to materials done.
"""
from __future__ import annotations

import time

from dotenv import load_dotenv

load_dotenv()

from growme.delta_report.node import run as run_delta  # noqa: E402
from growme.design_doc.node import run as run_design  # noqa: E402
from growme.miro_client import MiroClient  # noqa: E402
from growme.nudges import learner_fixtures  # noqa: E402
from growme.nudges.node import generate_for_session  # noqa: E402
from growme.research.node import run as run_research  # noqa: E402
from growme.schemas import WizardInputs  # noqa: E402
from growme.sessions.materials_node import run_all as run_materials  # noqa: E402
from growme.sessions.plan_node import run as run_plan  # noqa: E402


def main():
    inputs = WizardInputs(
        company_url="neon.tech",
        company_alias="Photon DB",
        audience_description="12 mid-market AEs, 1-3 yrs tenure, expanding into healthcare",
        selected_behavior_ids=[
            "pic_pbo_quantify_pain",
            "pic_rc_capabilities_outcomes",
            "pic_diff_differentiate",
        ],
    )

    t0 = time.time()
    enriched = run_research(inputs)
    t1 = time.time()
    print(f"[research] {t1-t0:.1f}s — {len(enriched.sources_used)} sources")

    doc = run_design(enriched, inputs)
    t2 = time.time()
    print(f"[design_doc] {t2-t1:.1f}s")

    plans = run_plan(doc.full_markdown, enriched, inputs)
    t3 = time.time()
    print(f"[session_plan] {t3-t2:.1f}s — {len(plans)} sessions")

    client = MiroClient()
    mats = run_materials(client, plans, enriched, inputs=inputs, design_doc=doc)
    t4 = time.time()
    print(f"[materials] {t4-t3:.1f}s — {len(mats)} session frames")

    all_responses = learner_fixtures.all_responses()
    nudges_total = []
    for s in (1, 2, 3, 4):
        nudges_total.extend(generate_for_session(client, s, all_responses, enriched))
    t5 = time.time()
    print(f"[nudges] {t5-t4:.1f}s — {len(nudges_total)} nudges")

    report = run_delta(all_responses, enriched, client)
    t6 = time.time()
    print(f"[delta] {t6-t5:.1f}s")

    print(f"\nTOTAL: {t6-t0:.1f}s")
    print(f"Movements: {[(m.behavior_id, f'{m.pct_moved_from_rarely_to_often:+.0%}') for m in report.behavior_movements]}")


if __name__ == "__main__":
    main()
